async function doExport() { const data = { app: 'EvilNote', exported_at: new Date().toISOString(), sections: sectionsCache, notes: notesCache }; const name = backupFileName(); const json = JSON.stringify(data, null, 2); if (!Storage.isNative) { const blob = new Blob([json], { type: 'application/octet-stream' }); const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = name; document.body.appendChild(a); a.click(); document.body.removeChild(a); showToast('✓ Файл ' + name + ' скачан'); return; } try { await Storage.Filesystem.writeFile({ path: name, data: json, directory: 'CACHE', encoding: 'utf8' }); const uri = await Storage.Filesystem.getUri({ path: name, directory: 'CACHE' }); if (window.Capacitor && window.Capacitor.Plugins && window.Capacitor.Plugins.Share && window.Capacitor.Plugins.Share.shareFiles) { await window.Capacitor.Plugins.Share.shareFiles({ files: [uri.uri], dialogTitle: 'Сохранить резервную копию' }); showToast('✓ Файл ' + name); } else if (navigator.share && navigator.canShare) { const file = new File([json], name, { type: 'application/json' }); if (navigator.canShare({ files: [file] })) { await navigator.share({ files: [file], title: 'EvilNote backup' }); showToast('✓ Файл ' + name); } else { throw new Error('Файлы не поддерживаются'); } } else { throw new Error('Нет доступного способа поделиться'); } } catch (e) { console.error('[Export error]', e); showToast('Ошибка экспорта: ' + (e.message || 'неизвестно'), true); } }
